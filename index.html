Especificación del Workflow (.github/workflows/main.yml)

Definición YAML ejecutada en GitHub Actions

name: Deploy to Surge

on:
  push:
    branches:
      - main
      - master
  workflow_dispatch:

jobs:
  deploy:
    runs-on: ubuntu-latest
    steps:
      - name: Checkout Code
        uses: actions/checkout@v4

      - name: Setup Node.js
        uses: actions/setup-node@v4
        with:
          node-version: 20

      - name: Install Surge
        run: npm install --global surge

      - name: Deploy to Surge.sh
        run: surge ./ nikkoban-primer-parcial.surge.sh --token ${{ secrets.SURGE_TOKEN }}
        env:
          SURGE_LOGIN: ${{ secrets.SURGE_LOGIN }}
          SURGE_TOKEN: ${{ secrets.SURGE_TOKEN }}
